\chapter*{Liste des abréviations}
\addcontentsline{toc}{chapter}{Liste des abréviations}
\frontmark{Liste des abréviations}

{\setstretch{1.1}\small\setlength{\parskip}{3pt}
\textbf{AIC} : critère d'information d'Akaike.

\textbf{API} : interface de programmation (\emph{Application Programming Interface}).

\textbf{AUC} : aire sous la courbe ROC.

\textbf{C-MAPSS} : simulateur de turboréacteur de la NASA (\emph{Commercial Modular Aero-Propulsion System Simulation}).

\textbf{FMDS} : fiabilité, maintenabilité, disponibilité, sécurité.

\textbf{HI} : indicateur de santé (\emph{health indicator}).

\textbf{HPC} : compresseur haute pression (\emph{high pressure compressor}).

\textbf{IC} : intervalle de confiance.

\textbf{KM} : estimateur de Kaplan-Meier.

\textbf{KS} : test de Kolmogorov-Smirnov.

\textbf{LSTM} : réseau de neurones récurrent à mémoire longue (\emph{Long Short-Term Memory}).

\textbf{MAE} : erreur absolue moyenne.

\textbf{MCO} : moindres carrés ordinaires.

\textbf{MDT / MUT} : durées moyennes d'indisponibilité et de disponibilité (\emph{mean down time}, \emph{mean up time}).

\textbf{MTBF} : durée moyenne entre deux défaillances (\emph{mean time between failures}).

\textbf{MTTF} : durée moyenne de fonctionnement avant défaillance (\emph{mean time to failure}).

\textbf{MTTR} : durée moyenne de réparation (\emph{mean time to repair}).

\textbf{MV} : maximum de vraisemblance.

\textbf{PHM} : surveillance de l'état de santé et pronostic (\emph{prognostics and health management}).

\textbf{RMSE} : racine de l'erreur quadratique moyenne.

\textbf{ROC} : courbe caractéristique de fonctionnement d'un classifieur.

\textbf{RUL} : durée de vie résiduelle (\emph{remaining useful life}).

\textbf{SVM} : machine à vecteurs de support.
}
